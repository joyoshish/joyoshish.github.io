import numpy as np
import matplotlib.pyplot as plt

# Function and gradient
def f(w):
    return (w - 3)**2

def grad_f(w):
    return 2 * (w - 3)

# Settings
eta = 0.1
w = 0
steps = 50

ws = [w]
fs = [f(w)]

for _ in range(steps):
    w = w - eta * grad_f(w)
    ws.append(w)
    fs.append(f(w))

plt.figure(figsize=(10, 6))
plt.plot(range(steps+1), fs, marker='o', label='$f(w)$')
plt.xlabel('Iteration')
plt.ylabel('Loss')
plt.title('Convergence of Gradient Descent on $f(w) = (w - 3)^2$')
plt.grid(True)
plt.legend()
plt.show()

import numpy as np
import plotly.graph_objects as go

# Grid
x = np.linspace(-2, 2, 50)
y = np.linspace(-2, 2, 50)
X, Y = np.meshgrid(x, y)

# True surface
Z_true = 3 + 2 * X + Y**2

# Linear approximation
Z_linear = 3 + 2 * X

# Create surface plots
surface_true = go.Surface(z=Z_true, x=x, y=y, colorscale='Viridis', opacity=0.9, name='True Function')
surface_linear = go.Surface(z=Z_linear, x=x, y=y, colorscale='Hot', opacity=0.6, name='Linear Approx')

fig = go.Figure(data=[surface_true, surface_linear])
fig.update_layout(
    title='True Function vs Linear Approximation',
    scene=dict(
        xaxis_title='x',
        yaxis_title='y',
        zaxis_title='f(x, y)'
    ),
    margin=dict(l=0, r=0, b=0, t=50)
)

# Save to HTML
fig.write_html("assets/plotly/linear_approx.html", include_plotlyjs='cdn')
